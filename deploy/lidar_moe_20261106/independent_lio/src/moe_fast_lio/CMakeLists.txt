cmake_minimum_required(VERSION 3.8)
project(moe_fast_lio)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_BUILD_TYPE Release)
add_compile_options(-O2 -g1 -pthread -fexceptions)
add_definitions(-DROOT_DIR="${CMAKE_CURRENT_SOURCE_DIR}/" -DMP_PROC_NUM=1)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(unitree_go REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(PCL REQUIRED COMPONENTS common io filters kdtree)
find_package(OpenMP REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME} "msg/Pose6D.msg")
ament_export_dependencies(rosidl_default_runtime)
add_executable(moe_fastlio_mapping src/laserMapping.cpp include/ikd-Tree/ikd_Tree.cpp src/preprocess.cpp)
target_include_directories(moe_fastlio_mapping PRIVATE include ${PCL_INCLUDE_DIRS})
target_link_libraries(moe_fastlio_mapping ${PCL_LIBRARIES} Eigen3::Eigen OpenMP::OpenMP_CXX)
rosidl_target_interfaces(moe_fastlio_mapping ${PROJECT_NAME} "rosidl_typesupport_cpp")
ament_target_dependencies(moe_fastlio_mapping rclcpp geometry_msgs nav_msgs sensor_msgs std_msgs std_srvs visualization_msgs tf2_ros pcl_conversions)
install(TARGETS moe_fastlio_mapping DESTINATION lib/${PROJECT_NAME})
add_executable(moe_body_imu_bridge src/body_imu_bridge.cpp)
ament_target_dependencies(moe_body_imu_bridge rclcpp sensor_msgs std_msgs unitree_go)
install(TARGETS moe_body_imu_bridge DESTINATION lib/${PROJECT_NAME})
ament_package()
